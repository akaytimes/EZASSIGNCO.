  <script>
    lucide.createIcons();

    let currentService = 'writing';
    let pages = 1;
    const rates = {
      writing: 10,  // £10 per 500 words (1 page)
      technical: 15, // £15 per 500 words (1 page)
      online: 15     // £15 per 500 words (1 page)
    };

    function setService(service) {
      currentService = service;
      document.querySelectorAll('.service-btn').forEach(btn => {
        btn.classList.remove('border-indigo-600', 'bg-indigo-50', 'text-indigo-700');
        btn.classList.add('border-slate-200', 'bg-white', 'text-slate-700');
      });

      const activeBtn = document.getElementById(`btn-${service}`);
      activeBtn.classList.remove('border-slate-200', 'bg-white', 'text-slate-700');
      activeBtn.classList.add('border-indigo-600', 'bg-indigo-50', 'text-indigo-700');

      calculatePrice();
    }

    function updatePages(delta) {
      pages = Math.max(1, pages + delta);
      document.getElementById('page-count').innerText = pages;
      document.getElementById('word-count').innerText = `${pages * 500} Words`;
      calculatePrice();
    }

    function calculatePrice() {
      const pricePerPage = rates[currentService];
      const totalPrice = pages * pricePerPage;
      document.getElementById('total-price').innerText = `£${totalPrice}`;

      const waMessage = encodeURIComponent(`Hi EzAssign, I need help with ${currentService} service for ${pages} page(s) (${pages * 500} words). Estimated price: £${totalPrice}.`);
      document.getElementById('wa-quote-btn').href = `https://wa.me/919256466594?text=${waMessage}`;
    }
  </script>
